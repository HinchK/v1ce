### `src/lib/app-params.js`
```js
const isNode = typeof window === 'undefined';
const windowObj = isNode ? { localStorage: new Map() } : window;
const storage = windowObj.localStorage;
const toSnakeCase = (str) => str.replace(/([A-Z])/g, '_$1').toLowerCase();
const getAppParamValue = (paramName, { defaultValue = undefined, removeFromUrl = false } = {}) => {
  if (isNode) return defaultValue;
  const storageKey = `base44_${toSnakeCase(paramName)}`;
  const urlParams = new URLSearchParams(window.location.search);
  const searchParam = urlParams.get(paramName);
  if (removeFromUrl) {
    urlParams.delete(paramName);
    const newUrl = `${window.location.pathname}${urlParams.toString() ? `?${urlParams.toString()}` : ""}${window.location.hash}`;
    window.history.replaceState({}, document.title, newUrl);
  }
  if (searchParam) { storage.setItem(storageKey, searchParam); return searchParam; }
  if (defaultValue) { storage.setItem(storageKey, defaultValue); return defaultValue; }
  const storedValue = storage.getItem(storageKey);
  return storedValue || null;
}
const getAppParams = () => {
  if (getAppParamValue("clear_access_token") === 'true') {
    storage.removeItem('base44_access_token'); storage.removeItem('token');
  }
  return {
    appId: getAppParamValue("app_id", { defaultValue: import.meta.env.VITE_BASE44_APP_ID }),
    token: getAppParamValue("access_token", { removeFromUrl: true }),
    fromUrl: getAppParamValue("from_url", { defaultValue: window.location.href }),
    functionsVersion: getAppParamValue("functions_version", { defaultValue: import.meta.env.VITE_BASE44_FUNCTIONS_VERSION }),
    appBaseUrl: getAppParamValue("app_base_url", { defaultValue: import.meta.env.VITE_BASE44_APP_BASE_URL }),
  }
}
export const appParams = { ...getAppParams() }
```

```jsx
import React, { createContext, useContext, useState, useEffect } from "react";
const CoinContext = createContext();
export function CoinProvider({ children }) {
  const [coinData, setCoinData] = useState({ days: 0, color: "#E0E0E0", displayName: "" });
  useEffect(() => {
    try {
      const profile = localStorage.getItem("v1ce_profile") || localStorage.getItem("v1ce_guest_profile");
      if (profile) {
        const parsed = JSON.parse(profile);
        setCoinData({ days: parsed.days || 0, color: parsed.coin_color || "#E0E0E0", displayName: parsed.display_name || "" });
      }
    } catch {}
  }, []);
  const updateCoinData = (updates) => setCoinData((prev) => ({ ...prev, ...updates }));
  return <CoinContext.Provider value={{ coinData, updateCoinData }}>{children}</CoinContext.Provider>;
}
export function useCoinContext() {
  const context = useContext(CoinContext);
  if (!context) throw new Error("useCoinContext must be used within CoinProvider");
  return context;
}
```

```jsx
import React, { createContext, useContext, useState, useEffect } from "react";
const ThemeContext = createContext({ dark: false, toggle: () => {} });
export function ThemeProvider({ children }) {
  const [dark, setDark] = useState(() => {
    try { return localStorage.getItem("v1ce-theme") === "dark"; } catch { return false; }
  });
  useEffect(() => {
    const root = document.documentElement;
    if (dark) root.classList.add("dark"); else root.classList.remove("dark");
    try { localStorage.setItem("v1ce-theme", dark ? "dark" : "light"); } catch {}
  }, [dark]);
  return <ThemeContext.Provider value={{ dark, toggle: () => setDark((d) => !d) }}>{children}</ThemeContext.Provider>;
}
export const useTheme = () => useContext(ThemeContext);
```


### `src/lib/PremiumContext.jsx`
```jsx
import React, { createContext, useContext, useState, useEffect } from "react";
const PremiumContext = createContext();
export function PremiumProvider({ children }) {
  const [isPremium, setIsPremium] = useState(false);
  useEffect(() => {
    try {
      const profile = localStorage.getItem("v1ce_profile") || localStorage.getItem("v1ce_guest_profile");
      if (profile) { const parsed = JSON.parse(profile); setIsPremium(!!parsed.is_premium); }
    } catch {}
  }, []);
  const unlockPremium = () => {
    setIsPremium(true);
    try {
      const raw = localStorage.getItem("v1ce_profile") || localStorage.getItem("v1ce_guest_profile");
      const existing = raw ? JSON.parse(raw) : {};
      localStorage.setItem("v1ce_profile", JSON.stringify({ ...existing, is_premium: true }));
    } catch {}
  };
  return <PremiumContext.Provider value={{ isPremium, unlockPremium }}>{children}</PremiumContext.Provider>;
}
export function usePremium() {
  const ctx = useContext(PremiumContext);
  if (!ctx) throw new Error("usePremium must be used within PremiumProvider");
  return ctx;
}
```

```jsx
import { Toaster } from "@/components/ui/toaster"
import { ThemeProvider } from "@/lib/ThemeContext"
import { CoinProvider } from "@/lib/CoinContext"
import { PremiumProvider } from "@/lib/PremiumContext"
import { QueryClientProvider } from '@tanstack/react-query'
import { queryClientInstance } from '@/lib/query-client'
import { BrowserRouter as Router, Route, Routes } from 'react-router-dom';
import { useState, useEffect } from 'react';
import PageNotFound from './lib/PageNotFound';
import { AuthProvider } from '@/lib/AuthContext';
import AppLayout from './components/layout/AppLayout';
import Home from './pages/Home';
import Customize from './pages/Customize';
import Widget from './pages/Widget';
import CoinWidget from './pages/CoinWidget';
import Analytics from './pages/Analytics';
import Profile from './pages/Profile';
import Lounge from './pages/Lounge';
import Game from './pages/Game';
import Premium from './pages/Premium';
import Friends from './pages/Friends';
import OnboardingFlow from './components/onboarding/OnboardingFlow';


const AuthenticatedApp = () => {
  const [hasProfile, setHasProfile] = useState(false);
  const [isChecking, setIsChecking] = useState(true);
  useEffect(() => {
    const profile = localStorage.getItem("v1ce_profile") || localStorage.getItem("v1ce_guest_profile");
    setHasProfile(!!profile);
    setIsChecking(false);
  }, []);
  if (isChecking) return <div className="flex items-center justify-center h-screen"><div className="w-5 h-5 border-2 border-foreground/20 border-t-foreground rounded-full animate-spin" /></div>;
  if (!hasProfile) return <OnboardingFlow onComplete={() => window.location.href = "/"} />;
  return (
    <Routes>
      <Route path="/coin-widget" element={<CoinWidget />} />
      <Route element={<AppLayout />}>
        <Route path="/" element={<Home />} />
        <Route path="/customize" element={<Customize />} />
        <Route path="/widget" element={<Widget />} />
        <Route path="/analytics" element={<Analytics />} />
        <Route path="/profile" element={<Profile />} />
        <Route path="/lounge" element={<Lounge />} />
        <Route path="/game" element={<Game />} />
        <Route path="/premium" element={<Premium />} />
        <Route path="/friends" element={<Friends />} />
      </Route>
      <Route path="*" element={<PageNotFound />} />
    </Routes>
  );
};


function App() {
  return (
    <QueryClientProvider client={queryClientInstance}>
      <Router>
        <AuthProvider>
          <ThemeProvider>
            <CoinProvider>
              <PremiumProvider>
                <AuthenticatedApp />
              </PremiumProvider>
            </CoinProvider>
          </ThemeProvider>
        </AuthProvider>
        <Toaster />
      </Router>
    </QueryClientProvider>
  )
}
export default App
```

```jsx
function formatDate(date) {
  const y = date.getUTCFullYear();
  const m = String(date.getUTCMonth() + 1).padStart(2, "0");
  const d = String(date.getUTCDate()).padStart(2, "0");
  return `${y}${m}${d}`;
}


function generateICS(sobrietyDate, displayName) {
  const start = new Date(sobrietyDate + "T00:00:00");
  const name = displayName || "You";


  const events = MILESTONES.map((m) => {
    const date = new Date(start);
    date.setDate(date.getDate() + m.days);
    const dateStr = formatDate(date);
    const endStr = formatDate(new Date(date.getTime() + 24 * 60 * 60 * 1000));
    const uid = `v1ce-milestone-${m.days}-${dateStr}@v1ce.app`;
    const stamp = formatDate(new Date()) + "T000000Z";


    return [
      "BEGIN:VEVENT",
      `UID:${uid}`,
      `DTSTAMP:${stamp}`,
      `DTSTART;VALUE=DATE:${dateStr}`,
      `DTEND;VALUE=DATE:${endStr}`,
      `SUMMARY:🎉 ${m.label} Sober!`,
      `DESCRIPTION:${name} reached ${m.label} of sobriety. ${m.message}`,
      "BEGIN:VALARM",
      "TRIGGER:PT9H",
      "ACTION:DISPLAY",
      `DESCRIPTION:🎉 ${m.label} Sober!`,
      "END:VALARM",
      "END:VEVENT",

  }).join("\r\n");

    "VERSION:2.0",
    "PRODID:-//V1CE//Sobriety Milestones//EN",
```

Here's the rest of the advice I'd give a newcomer who just showed up and wants to run this thing on their own:


---


**1. Get it running locally first — before touching anything.**
- Install Node (use the LTS version, not latest).
- `npm install` in the project root, then `npm run dev`. Vite will spin up on localhost.
- Don't start changing code until it boots clean and you can click around. If it errors immediately, it's almost certainly the backend SDK calls failing — that's expected and that's the first thing you'll deal with.


**2. The backend is the whole game.**
This app isn't just a frontend — it talks to a hosted backend for *everything*: login, the database (sobriety profiles, friends, blocked users, chat messages, giveaway entries), file uploads, the LLM, email, Stripe checkout. The moment you leave the platform, every one of those calls is a dead endpoint. You'll see `base44.entities...`, `base44.auth...`, `base44.integrations...` all over the code. That's your map of what to replace.


**3. Pick a replacement backend before writing any code.** The fastest options for a newcomer:
- **Supabase** — Postgres + auth + file storage + realtime, free tier, JS SDK that's shaped similarly. Closest 1:1 swap.
- **Firebase** — NoSQL, auth, storage, push. Good if you like document-style data.
- **PocketBase** — single binary, self-hosted, dead simple. Great for "just messing around."


Don't try to build your own API from scratch yet — that's a rabbit hole. Use one of these and swap the SDK calls.


**4. Stub first, build second.**
You don't need to replace everything at once. Comment out / mock the backend calls and get the UI fully working with fake data. Then wire real calls one entity at a time. This keeps you from staring at a broken app for a week.


**5. The data model is already documented for you.**
The entity schemas (SobrietyProfile, FriendConnection, BlockedUser, LoungeChatMessage, GiveawayEntry) are basically your database blueprint. Recreate those as tables/collections in whatever backend you pick. You're not designing from scratch — you're translating.


**6. Environment variables.**
The app reads app params from `src/lib/app-params.js` and secrets from the platform. Off-platform, you'll move those into a `.env` file (Vite uses `VITE_`-prefixed vars) and a real secret manager for anything server-side. Don't hardcode keys in frontend code — Stripe secret keys, LLM API keys, etc. belong in a backend you control.


**7. Mobile is the last step, not the first.**
Don't open Expo on day one. Get the web app running → replace the backend → get it fully working in the browser → *then* decide Capacitor (wrap the web app, easy) vs. Expo (rewrite in React Native, hard). A newcomer messing around should almost certainly start with Capacitor.


**8. Expect the build to be opinionated.**
The project has Tailwind config, shadcn/ui components, a specific font setup, and token-based theming. All of that is standard and portable — it'll work anywhere Vite runs. The *only* platform-specific bits are the `@base44/sdk` import and the `@base44/vite-plugin`. Strip those two and you're 90% portable.


**9. Version control immediately.**
`git init` → commit the working state *before* you start ripping out the backend. You'll want to diff against "the version that worked" constantly.


**10. Don't try to replicate everything at once.**
The app has a lot — friends, lounge chat, a game, giveaways, birthday stuff, gifting, premium. Pick *one* feature (the sobriety counter + coin on the Home page) and make that fully work standalone. That's your proof-of-concept. Everything else is a repeat of the same pattern.

A few last things I'd add, then I'd let them go build:


---


**11. The onboarding gate will trip you up.**
The app checks `localStorage` for a profile and redirects to an onboarding flow if it's missing. When you first run it standalone, you'll likely land on onboarding or a blank screen because the profile save calls a dead backend. Either mock that flow or bypass it temporarily so you can actually see the app while you work.


**12. Realtime subscriptions are a gotcha.**
Several components use `base44.entities.X.subscribe(...)` for live updates (chat messages, friend lists). Supabase and Firebase both have realtime, but the API shape is different — it's not a drop-in. Budget time to rewire those; they won't "just work" after a find-and-replace.


**13. Row-level security doesn't move with you.**
The entity schemas have `rls` rules (who can read/edit whose records). That logic lives in the platform, not your code. When you recreate the tables, you'll need to re-implement those policies in your new backend's security rules (Supabase RLS, Firebase rules, etc.) — or your app will be either locked down too tight or wide open. Don't skip this; it's how one user's sobriety profile stays private from another.


**14. Stripe is its own migration.**
The checkout flow uses a backend function that creates Stripe sessions. Off-platform you'll run that server-side yourself (a small Node/Express endpoint or a serverless function) with your own Stripe secret key. The frontend `@stripe/react-stripe-js` part stays — it's just the session-creation call that needs a new home.


**15. Don't trust the localStorage-only paths.**
A lot of the app falls back to `localStorage` for guest/unauthenticated users. That's fine for prototyping, but it means data lives only on one device. When you wire a real backend, decide whether guests stay localStorage-only or you give them a real anonymous account. It affects the whole data model.


**16. The i18n file is huge and that's a good thing.**
All five languages are already there and self-contained. That'll port over with zero changes — one less thing to worry about.


**17. One thing I'd genuinely warn them about: don't half-leave.**
The worst spot to be in is *some* calls hitting the old platform and *some* hitting your new backend. Either fully replace the SDK or fully stay — a hybrid will break auth and data consistency in confusing ways. Pick a cutover point and commit.

```jsx
            const p = profiles[0];
            // Mirror to localStorage so future loads are instant
            saveLocalProfile(p);
            setProfile(p);
            setSobrietyDate(p.sobriety_date || "");
            setSubstances(p.substances || []);
          }
        }
      } catch {
        // silently ignore — no profile found
      }


      setIsLoading(false);
    }
    loadProfile();
  }, []);






  const handleSaveDate = (newDate) => {
    setSobrietyDate(newDate);
    saveLocalProfile({ sobriety_date: newDate });
    base44.auth.isAuthenticated().then((authed) => {
      if (authed && profile?.id) {
        base44.entities.SobrietyProfile.update(profile.id, { sobriety_date: newDate }).catch(() => {});
      }
    });
  };


  const handleSubstanceChange = (newSubstances) => {
    setSubstances(newSubstances);
    saveLocalProfile({ substances: newSubstances });
    base44.auth.isAuthenticated().then((authed) => {
      if (authed && profile?.id) {
        base44.entities.SobrietyProfile.update(profile.id, { substances: newSubstances }).catch(() => {});
      }
    });
  };


  const daysSober = sobrietyDate
    ? Math.max(0, Math.floor((new Date() - new Date(sobrietyDate)) / (1000 * 60 * 60 * 24)))
    : 0;


  // Sync days to global state whenever they change
  useEffect(() => {
    updateCoinData({ days: daysSober });
  }, [daysSober, updateCoinData]);


  if (isLoading) {
    return (
      <div className="flex items-center justify-center py-24">
        <div className="w-5 h-5 border-2 border-foreground/20 border-t-foreground rounded-full animate-spin" />
      </div>
    );
  }


  // ── DASHBOARD ────────────────────────────────────────────────────
  return (
    <div className="pb-16">


      {/* ── HERO ── */}
      <section className="px-5 pt-6 pb-10 border-b-2 border-foreground relative overflow-hidden">
        <Starburst size={64} className="absolute top-4 right-4 pointer-events-none opacity-[0.06]" />
        <BlobSplat size={90} className="absolute -bottom-4 -left-6 pointer-events-none opacity-[0.05]" />
        <motion.div initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.4 }}>
          <div className="mb-6">
            <p
              className="font-display text-foreground leading-none"
              style={{ fontSize: "clamp(6rem, 28vw, 9rem)", letterSpacing: "-0.01em", lineHeight: 0.9 }}
            >
              {daysSober}
            </p>
            <p className="font-display text-3xl mt-2 tracking-wider">
              <RotatingLabel />
            </p>
          </div>


          <div className="flex justify-center py-4" style={{ width: "100%" }}>
            <SobrietyCoin
              days={daysSober}
              shape={profile?.coin_shape || "circle"}
              color={profile?.coin_color || "#F5D680"}
              numberStyle={profile?.number_style || "classic"}
              substances={substances}
              displayName={profile?.display_name}
              motto={profile?.coin_motto}
              customShapePath={profile?.coin_shape_path || ""}
              showBorder={profile?.coin_show_border !== false}
              coinPhoto={profile?.coin_photo || ""}
              borderColor={profile?.coin_border_color || ""}
              numberColor={profile?.coin_number_color || ""}
              imageOnlyMode={profile?.coin_image_only || false}
            />
          </div>


          <div className="flex justify-center mt-4">
            <Link
              to="/customize"
              className="text-xs font-body tracking-widest uppercase text-muted-foreground hover:text-foreground transition-colors border-b border-muted-foreground hover:border-foreground"
            >
              {t("home.customize")}
            </Link>
          </div>
        </motion.div>
      </section>


      {/* ── COUNTER ── */}
      <section className="px-5 py-10 border-b-2 border-foreground relative overflow-hidden">
        <DiamondGrid size={72} className="absolute top-4 right-3 pointer-events-none opacity-[0.08]" />
        <p className="font-display text-4xl text-foreground mb-6 tracking-wider">{t("home.timeElapsed").split("\n").map((line, i) => <React.Fragment key={i}>{line}{i === 0 && <br />}</React.Fragment>)}</p>
        <SobrietyCounter sobrietyDate={sobrietyDate} />
      </section>


      {/* ── SOBER SINCE ── */}
      <section className="px-5 py-10 border-b-2 border-foreground relative overflow-hidden">
        <p className="font-body text-[10px] font-700 tracking-[0.3em] uppercase text-muted-foreground mb-1">{t("home.soberSince")}</p>
        <div className="flex items-center justify-between gap-4">
          <p className="font-display text-3xl text-foreground tracking-wide">
            {new Date(sobrietyDate).toLocaleDateString("en-US", {
              month: "long", day: "numeric", year: "numeric",
            }).toUpperCase()}
          </p>
          <Input
            type="date"
            value={sobrietyDate}
            onChange={(e) => handleSaveDate(e.target.value)}
            className="w-auto font-body text-xs bg-white border-2 border-foreground rounded-none focus-visible:ring-0"
            max={new Date().toISOString().split("T")[0]}
          />
        </div>
      </section>


      {/* ── SUBSTANCES ── */}
      <section className="px-5 py-10 border-b-2 border-foreground relative overflow-hidden">
        <Crosshair size={60} className="absolute top-4 right-4 pointer-events-none opacity-[0.08]" />
        <Halftone size={56} className="absolute bottom-4 right-0 pointer-events-none opacity-[0.06]" />
        <p className="font-display text-foreground leading-none mb-6" style={{ fontSize: "clamp(2.8rem, 14vw, 4rem)" }}>
          {t("home.whatsYourDoc").split("\n").map((line, i) => <React.Fragment key={i}>{line}{i === 0 && <br />}</React.Fragment>)}
        </p>
        <SubstanceChecklist selected={substances} onChange={handleSubstanceChange} />
      </section>


      {/* ── MILESTONES ── */}
      {daysSober > 0 && (
        <section className="px-5 py-10 border-b-2 border-foreground relative overflow-hidden">
          <AsteriskStar size={48} className="absolute top-5 right-5 pointer-events-none opacity-[0.09]" />
          <WarpedTorus size={100} className="absolute -bottom-4 -right-4 pointer-events-none opacity-[0.07]" />
          <p className="font-display text-foreground leading-none mb-6" style={{ fontSize: "clamp(2.8rem, 14vw, 4rem)" }}>
            {t("home.yourMilestones").split("\n").map((line, i) => <React.Fragment key={i}>{line}{i === 0 && <br />}</React.Fragment>)}
          </p>
          <MilestoneTimeline days={daysSober} />
          <div className="mt-6">
            <MilestoneCalendarExport sobrietyDate={sobrietyDate} displayName={profile?.display_name} />
          </div>
        </section>
      )}


    </div>
  );
}
```
